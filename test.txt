import os
import requests
import numpy as np
import pandas as pd
import time
from datetime import datetime
from concurrent.futures import ThreadPoolExecutor, as_completed
from urllib.parse import urljoin
import re

# === Adapter untuk multiple source IP ===
class SourceIPAdapter(requests.adapters.HTTPAdapter):
    def __init__(self, source_address, **kwargs):
        self.source_address = source_address
        super(SourceIPAdapter, self).__init__(**kwargs)

    def init_poolmanager(self, *args, **kwargs):
        kwargs['source_address'] = (self.source_address, 0)
        super(SourceIPAdapter, self).init_poolmanager(*args, **kwargs)

# === Zipf Distribution ===
def zipf_mandelbrot(N, q, s):
    ranks = np.arange(1, N + 1)
    weights = (ranks + q) ** -s
    return weights / weights.sum()

# === Deteksi jenis konten dari URL ===
def get_content_type_from_url(url):
    if re.search(r'\.(jpg|jpeg|png|gif|bmp|svg|webp)$', url, re.IGNORECASE):
        return 'image'
    elif re.search(r'\.(html|htm|php|asp|aspx|jsp)$', url, re.IGNORECASE):
        return 'text'
    else:
        return 'other'

# === Ambil link resource dari HTML ===
def extract_links(html, base_url):
    pattern = r'<img[^>]+src=["\'](.*?)["\']|<script[^>]+src=["\'](.*?)["\']|<link[^>]+href=["\'](.*?)["\']'
    matches = re.findall(pattern, html, re.IGNORECASE)
    links = set()
    for match in matches:
        for link in match:
            if link:
                absolute_link = urljoin(base_url, link)
                links.add(absolute_link)
    return links

# === Fetch resource (return size + jenis) ===
def fetch_url(session, url):
    try:
        response = session.get(url, timeout=5)
        size = len(response.content)
        content_type = get_content_type_from_url(url)
        return size, content_type
    except requests.exceptions.RequestException:
        return 0, 'error'

# === Single HTTP request & link tracing ===
def make_request(url, results, session):
    try:
        rtt_start = time.time()
        response = session.get(url, timeout=5)
        rtt = (time.time() - rtt_start) * 1000
        html = response.text
        status_code = response.status_code

        links = extract_links(html, url)
        total_size = len(response.content)
        total_image_size = 0
        total_text_size = 0

        fetch_start = time.time()
        with ThreadPoolExecutor(max_workers=10) as executor:
            future_to_url = {executor.submit(fetch_url, session, link): link for link in links}
            for future in as_completed(future_to_url):
                size, content_type = future.result()
                total_size += size
                if content_type == 'image':
                    total_image_size += size
                elif content_type == 'text':
                    total_text_size += size

        fetch_time = time.time() - fetch_start
        throughput = (total_size / 1024) / fetch_time if fetch_time > 0 else 0

        log_data = [
            url, round(rtt, 2), status_code,
            round(total_size / 1024, 2),
            round(throughput, 2),
            round(total_image_size / 1024, 2),
            round(total_text_size / 1024, 2)
        ]
        results.append(log_data)

        print(f"✅ {url} | RTT: {rtt:.2f} ms | Size: {total_size/1024:.2f} KB | TP: {throughput:.2f} KB/s | Img: {total_image_size/1024:.2f} KB | Txt: {total_text_size/1024:.2f} KB")

    except requests.exceptions.RequestException as e:
        log_data = [url, 0, f"Failed: {e}", 0, 0, 0, 0]
        results.append(log_data)
        print(f"❌ {url} failed: {e}")

# === Generate traffic menggunakan distribusi Zipf ===
def generate_traffic(urls, num_requests, rps, zipf_params, source_ips):
    probs = zipf_mandelbrot(len(urls), *zipf_params)
    results = []
    executor = ThreadPoolExecutor(max_workers=100)

    sessions = [requests.Session() for _ in source_ips]
    for session, ip in zip(sessions, source_ips):
        session.mount('http://', SourceIPAdapter(ip))

    for i in range(num_requests):
        url = np.random.choice(urls, p=probs)
        session = np.random.choice(sessions)
        executor.submit(make_request, url, results, session)
        time.sleep(1 / rps)

    executor.shutdown(wait=True)
    return results

# === Simpan hasil ke CSV ===
def log_to_csv(data_rows, filename='request_log_http.csv'):
    columns = [
        "URL", "RTT (ms)", "Status Code",
        "Total Size (KB)", "Throughput (KB/s)",
        "Image Size (KB)", "Text Size (KB)"
    ]
    df = pd.DataFrame(data_rows, columns=columns)
    df.to_csv(filename, index=False)

# === File selector ===
def list_csv_files():
    files = [f for f in os.listdir() if f.endswith('.csv')]
    if not files:
        print("Tidak ada file CSV!")
        exit()

    print("\n===== Pilih File CSV =====")
    for idx, file in enumerate(files):
        print(f"{idx + 1}. {file}")

    while True:
        try:
            choice = int(input("Masukkan nomor file: ")) - 1
            if 0 <= choice < len(files):
                return files[choice]
        except ValueError:
            pass
        print("Input tidak valid!")

# === Main Program ===
def main():
    print("############ TRAFFIC SIMULATOR ############")

    csv_file = list_csv_files()
    df = pd.read_csv(csv_file)
    urls = df['URL'].tolist()

    print("\n===== Parameter Traffic =====")
    num_urls = int(input("Jumlah URL yang digunakan (-url): "))
    num_requests = int(input("Jumlah total request (-req): "))
    rps = float(input("Request per detik (-rps): "))
    zipf_q = float(input("Zipf parameter q: "))
    zipf_s = float(input("Zipf parameter s: "))
    source_ips = input("Source IPs (pisah koma): ").split(',')

    results = generate_traffic(urls[:num_urls], num_requests, rps, (zipf_q, zipf_s), source_ips)
    log_to_csv(results)

    print("\n✅ Log disimpan ke: `request_log_http.csv`")

if __name__ == "__main__":
    main()
